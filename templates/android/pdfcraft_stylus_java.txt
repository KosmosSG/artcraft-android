    private volatile boolean pdfWheelEnabled = false;
    private final ConcurrentLinkedQueue<String> pdfWheelKeys = new ConcurrentLinkedQueue<>();
    private long pdfWheelLastUp, pdfWheelLastDown;
    public String pdfStylusGetLayout() {
        return getSharedPreferences("pdf_stylus_wheel", MODE_PRIVATE)
            .getString("layout", "0,1,2,3,4,5,6,7");
    }
    public void pdfStylusSetLayout(String layout) {
        if (layout != null && layout.length() <= 128)
            getSharedPreferences("pdf_stylus_wheel", MODE_PRIVATE)
                .edit().putString("layout", layout).apply();
    }
    public boolean pdfStylusEnabled() { return pdfWheelEnabled; }
    public void pdfStylusSetEnabled(boolean enabled) {
        pdfWheelEnabled = enabled;
        pdfWheelKeys.clear();
        pdfWheelLastUp = pdfWheelLastDown = 0L;
        getSharedPreferences("pdf_stylus_wheel", MODE_PRIVATE).edit().putBoolean("enabled", enabled).apply();
    }
    public String pdfStylusPoll() { return pdfWheelKeys.poll(); }
    public String pdfStylusDeviceIds() {
        StringBuilder ids = new StringBuilder();
        for (int id : android.view.InputDevice.getDeviceIds()) {
            android.view.InputDevice input = android.view.InputDevice.getDevice(id);
            if (input != null && input.getName() != null && input.getName().contains("Magic-Pencil 4s")) {
                if (ids.length() > 0) ids.append(',');
                ids.append(id);
            }
        }
        return ids.toString();
    }
    private boolean pdfStylusIntercept(KeyEvent event) {
        if (!pdfWheelEnabled) return false;
        android.view.InputDevice d = android.view.InputDevice.getDevice(event.getDeviceId());
        if (d == null || d.getName() == null || !d.getName().contains("Magic-Pencil 4s")) return false;
        int code = event.getKeyCode();
        if (code != KeyEvent.KEYCODE_DPAD_UP && code != KeyEvent.KEYCODE_DPAD_DOWN) return false;
        if (event.getAction() == KeyEvent.ACTION_UP && event.getRepeatCount() == 0) {
            long now = android.os.SystemClock.uptimeMillis();
            long last = code == KeyEvent.KEYCODE_DPAD_UP ? pdfWheelLastUp : pdfWheelLastDown;
            if (now - last >= 650L && pdfWheelKeys.size() < 12) {
                pdfWheelKeys.offer(code == KeyEvent.KEYCODE_DPAD_UP ? "D" : "S");
                if (code == KeyEvent.KEYCODE_DPAD_UP) pdfWheelLastUp = now; else pdfWheelLastDown = now;
                getWindow().getDecorView().postInvalidate();
            }
        }
        return true;
    }
// Passive stylus-only hover: never consume, click, draw or persist positions.
// NativeActivity can deliver pre-contact hover to Activity rather than SurfaceView.
// Observe both dispatch routes, including on a fresh launch before the first tap.
private volatile String pdfPenHover = "";
// The rendered native surface is the origin used by Android's native event queue.
private volatile View pdfPenSurface = null;
public boolean pdfStylusWindowFocused() {
    return getWindow() != null && getWindow().getDecorView() != null
        && getWindow().getDecorView().hasWindowFocus();
}
private void pdfObservePen(android.view.MotionEvent event, View localView) {
    if (event == null || event.getPointerCount() == 0) return;
    int tool = event.getToolType(0);
    // Ignore unrelated mouse/trackpad/touch pointer motion; do not steal input.
    if (tool != android.view.MotionEvent.TOOL_TYPE_STYLUS
            && tool != android.view.MotionEvent.TOOL_TYPE_ERASER) return;
    int action = event.getActionMasked();
    if (action == android.view.MotionEvent.ACTION_HOVER_ENTER
            || action == android.view.MotionEvent.ACTION_HOVER_MOVE) {
        float x = event.getX(), y = event.getY();
        // All paths must end in native SurfaceView-local physical pixels.
        // Listener events are local to their originating view. Activity
        // dispatch is window-local. Convert via window position, then subtract
        // the native rendering surface offset exactly once.
        if (localView != null) {
            int[] origin = new int[2];
            localView.getLocationInWindow(origin);
            x += origin[0];
            y += origin[1];
        }
        View surface = pdfPenSurface;
        if (surface != null && surface.isAttachedToWindow()) {
            int[] origin = new int[2];
            surface.getLocationInWindow(origin);
            x -= origin[0];
            y -= origin[1];
        }
        if (Float.isFinite(x) && Float.isFinite(y)) {
            pdfPenHover = x + "," + y;
            // Wakes the view-side hover path without changing tool selection.
            // egui separately polls at idle cadence when the wheel is enabled.
            getWindow().getDecorView().postInvalidateOnAnimation();
        } else {
            pdfPenHover = "";
        }
    } else if (action == android.view.MotionEvent.ACTION_HOVER_EXIT
            || action == android.view.MotionEvent.ACTION_DOWN
            || action == android.view.MotionEvent.ACTION_UP
            || action == android.view.MotionEvent.ACTION_CANCEL) {
        // Contact never counts as hover and must clear the previous location.
        pdfPenHover = "";
    }
}
private void pdfObservePen(android.view.MotionEvent event) {
    pdfObservePen(event, null);
}
public String pdfStylusHover() { return pdfPenHover; }
// Pre-contact generic hover is handled by the existing PDFCraft Activity override.
@Override public boolean dispatchTouchEvent(android.view.MotionEvent event) {
    pdfObservePen(event);
    return super.dispatchTouchEvent(event);
}
private void pdfInstallHover(View view) {
    // Decor callback provides an early passive fallback before the native
    // SurfaceView has received its first touch. Never consume the motion.
    if (view == getWindow().getDecorView()) {
        view.setOnHoverListener((v, event) -> { pdfObservePen(event, v); return false; });
    }
    if (view instanceof android.view.SurfaceView) {
        pdfPenSurface = view;
        view.setOnHoverListener((v, event) -> { pdfObservePen(event, v); return false; });
        view.setOnTouchListener((v, event) -> { pdfObservePen(event, v); return false; });
    } else if (view instanceof android.view.ViewGroup) {
        android.view.ViewGroup group = (android.view.ViewGroup) view;
        for (int i = 0; i < group.getChildCount(); i++) pdfInstallHover(group.getChildAt(i));
    }
}
@Override public boolean dispatchKeyEvent(KeyEvent event) {
    if (pdfStylusIntercept(event)) return true;
    return super.dispatchKeyEvent(event);
}
