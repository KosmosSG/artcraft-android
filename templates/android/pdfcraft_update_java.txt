// PDFCraft: user-initiated verified APK download and SAF export.
// No installer permission, automatic installation, or system installer hand-off.
private static final int PDF_UPDATE_SAVE_REQUEST = 0x5353;
private static final long PDF_UPDATE_MAX_BYTES = 400L * 1024L * 1024L;
private volatile File pdfDownloadedApk;
private volatile String pdfDownloadedApkName;

private void pdfUpdateNotice(String message, boolean ready) {
    try {
        JSONObject result = new JSONObject();
        result.put("kind", "download");
        result.put("message", message);
        result.put("ready", ready);
        pdfUpdateResults.offer(result.toString());
        pdfUpdateToast(message);
    } catch (Exception error) {
        Log.e(TAG, "PDFCraft update status unavailable", error);
    }
}
private static String pdfApprovedAssetName(String url, String digest) {
    final String prefix = "https://github.com/KosmosSG/artcraft-android/releases/download/";
    if (url == null || !url.startsWith(prefix) || digest == null
            || !digest.matches("sha256:[a-fA-F0-9]{64}")) return null;
    String[] parts = url.substring(prefix.length()).split("/", -1);
    if (parts.length != 2 || !parts[0].matches("pdfcraft-v[0-9]+\\.[0-9]+\\.[0-9]+-android\\.[0-9]+"))
        return null;
    String version = parts[0].substring("pdfcraft-v".length()).split("-android\\.")[0];
    String name = "PDFCraft-v" + version + "-arm64-release.apk";
    return name.equals(parts[1]) ? name : null;
}
public synchronized void pdfDownloadUpdate(String url, String sha256Digest) {
    final String apkName = pdfApprovedAssetName(url, sha256Digest);
    if (!"com.artcraft.pdfcraft".equals(getPackageName()) || apkName == null) {
        pdfUpdateNotice("Cannot download: invalid PDFCraft asset, package, or published SHA-256.", false);
        return;
    }
    // Deliberately do not download until the user explicitly selects Download APK.
    pdfBackground.execute(() -> {
        File temp = null;
        java.net.HttpURLConnection conn = null;
        try {
            temp = File.createTempFile("pdfcraft-verified-", ".apk", getCacheDir());
            java.net.URL address = new java.net.URL(url);
            conn = (java.net.HttpURLConnection) address.openConnection();
            conn.setConnectTimeout(10000);
            conn.setReadTimeout(15000);
            conn.setInstanceFollowRedirects(true);
            conn.setRequestProperty("User-Agent", "PDFCraft-Android-Verified-Download");
            if (conn.getResponseCode() != 200
                    || !"https".equalsIgnoreCase(conn.getURL().getProtocol()))
                throw new java.io.IOException("HTTPS asset download failed");
            long advertised = conn.getContentLengthLong();
            if (advertised > PDF_UPDATE_MAX_BYTES)
                throw new java.io.IOException("APK exceeds permitted download size");
            java.security.MessageDigest digest =
                    java.security.MessageDigest.getInstance("SHA-256");
            long total = 0L;
            try (InputStream input = conn.getInputStream();
                 FileOutputStream output = new FileOutputStream(temp)) {
                byte[] bytes = new byte[65536];
                for (int n; (n = input.read(bytes)) != -1; ) {
                    total += n;
                    if (total > PDF_UPDATE_MAX_BYTES)
                        throw new java.io.IOException("APK exceeds permitted download size");
                    digest.update(bytes, 0, n);
                    output.write(bytes, 0, n);
                }
                output.getFD().sync();
            }
            if (total < 1024)
                throw new java.io.IOException("Downloaded asset is not a valid APK size");
            StringBuilder actual = new StringBuilder("sha256:");
            for (byte value : digest.digest())
                actual.append(String.format(java.util.Locale.ROOT, "%02x", value & 255));
            if (!actual.toString().equalsIgnoreCase(sha256Digest))
                throw new SecurityException("Published GitHub SHA-256 mismatch");
            synchronized (this) {
                File previous = pdfDownloadedApk;
                pdfDownloadedApk = temp;
                pdfDownloadedApkName = apkName;
                if (previous != null && !previous.equals(temp)) previous.delete();
            }
            temp = null; // Ownership transferred only after digest validation.
            pdfUpdateNotice("Verified PDFCraft APK. Choose Save APK to export it.", true);
        } catch (Exception error) {
            pdfUpdateNotice("Cannot download verified APK: " + error.getMessage(), false);
        } finally {
            if (conn != null) conn.disconnect();
            if (temp != null) temp.delete();
        }
    });
}
private String pdfDownloadStatus() { return null; }

public void pdfExportDownloadedUpdate() {
    final File file = pdfDownloadedApk;
    final String name = pdfDownloadedApkName;
    if (file == null || !file.isFile() || name == null) {
        pdfUpdateNotice("Cannot save: download and verify the APK first.", false);
        return;
    }
    runOnUiThread(() -> {
        try {
            Intent intent = new Intent(Intent.ACTION_CREATE_DOCUMENT);
            intent.addCategory(Intent.CATEGORY_OPENABLE);
            intent.setType("application/vnd.android.package-archive");
            intent.putExtra(Intent.EXTRA_TITLE, name);
            startActivityForResult(intent, PDF_UPDATE_SAVE_REQUEST);
        } catch (Exception error) {
            pdfUpdateNotice("Cannot open Android Save As: " + error.getMessage(), true);
        }
    });
}
private boolean pdfHandleExportResult(int requestCode, int resultCode, Intent data) {
    if (requestCode != PDF_UPDATE_SAVE_REQUEST) return false;
    if (resultCode == RESULT_OK && data != null && data.getData() != null) {
        final android.net.Uri outputUri = data.getData();
        final File verified = pdfDownloadedApk;
        if (verified == null || !verified.isFile()) {
            pdfUpdateNotice("Cannot save: verified APK is no longer available.", false);
            return true;
        }
        pdfBackground.execute(() -> {
            try (InputStream in = new FileInputStream(verified);
                 java.io.OutputStream out = getContentResolver().openOutputStream(outputUri, "w")) {
                if (out == null) throw new java.io.IOException("Document provider refused output");
                byte[] bytes = new byte[65536];
                for (int n; (n = in.read(bytes)) != -1; ) out.write(bytes, 0, n);
                out.flush();
                pdfUpdateNotice("APK saved. Open it from Files to request Android installation.", true);
            } catch (Exception error) {
                pdfUpdateNotice("Cannot save APK: " + error.getMessage(), true);
            }
        });
    } else {
        pdfUpdateNotice("Save APK cancelled; verified download is still available.", true);
    }
    return true;
}
public synchronized void pdfInstallDownloadedUpdate() {
    pdfUpdateNotice("In-app APK installation is disabled. Android installation requires an explicit user action.", false);
}
private boolean pdfHandleInstallResult(Intent intent) { return false; }
